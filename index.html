async function installViaBot(it){ if (!it.file) { toast("Файл не указан"); return; } const tg = window.Telegram && window.Telegram.WebApp; if (!tg) { toast("Открой через Telegram"); return; } const userId = tg.initDataUnsafe && tg.initDataUnsafe.user && tg.initDataUnsafe.user.id; if (!userId) { toast("Не удалось получить ID"); return; } $("#modal").classList.remove("show"); const sp = document.getElementById("installSplash"); sp.classList.add("show"); let ok = false; try { const res = await fetch("https://fonarikkapp-bot.onrender.com/install", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ user_id: userId, file: it.file, name: it.name || "Файл", type: it.type || "plugin" }) }); const data = await res.json(); ok = data.ok === true; if (!ok) console.error("Server error:", data.error); } catch(e) { console.error("Fetch error:", e); } setTimeout(() => { sp.classList.remove("show"); setTimeout(() => { try { tg.close(); } catch(e) {} }, 300); }, 2000); }
